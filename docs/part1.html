1. coin.hh — Coin Class

cppcoin.hh
#ifndef COIN_HH
#define COIN_HH

#include <vector>
#include <cmath>        // sin()
#include "geometry.hh"
#include "window.hh"
#include "utils.hh"

class Coin {
private:
    static const std::vector<std::vector<int>> coin_sprite_;
    pro2::Pt pos_;
    double   initial_y;
    bool     collected_;

public:
    Coin(pro2::Pt pos) : pos_(pos), collected_(false), initial_y(pos.y) {}

    void update(int frame_count) {
        pos_.y = initial_y + (3 * sin(frame_count) * 0.001);
    }

    void collect() { collected_ = true; }

    void paint_coin(pro2::Window& window) const {
        pro2::paint_sprite(window, pos_, coin_sprite_, false);
    }

    bool is_collected() const { return collected_; }

    pro2::Rect get_rect() const {
        // {left, top, right, bottom}
        return {pos_.x - 4, pos_.y - 4, pos_.x + 4, pos_.y + 4};
    }
};

#endif

2. coin.cc — Sprite Pixel Data

The sprite is an 8×8 pixel coin. Color constants are file-local.

cppcoin.cc
#include <vector>
#include "window.hh"
#include "coin.hh"

const int b = pro2::black;
const int m = 0x5c4033;    // dark brown (shading)
const int y = pro2::yellow;
const int _ = 0x5c94fc;    // sky blue (transparent)

const std::vector<std::vector<int>> Coin::coin_sprite_ = {
    {_, _, b, b, b, b, _, _},
    {_, b, y, y, y, y, b, _},
    {b, y, y, b, m, y, y, b},
    {b, y, y, b, m, y, y, b},
    {b, y, y, b, m, y, y, b},
    {b, y, y, b, m, y, y, b},
    {_, b, y, y, y, y, b, _},
    {_, _, b, b, b, b, _, _},
};
Note: The static member must be defined in exactly one .cc file using the Coin:: scope qualifier, otherwise the linker will produce an undefined reference error.

4. geometry.hh — Collision Detection

Uses the separating axis approach: two rectangles overlap unless separated on any axis.

diffgeometry.hh
  struct Rect {
      int left, top, right, bottom;
  };

+ inline bool overlaps(const Rect& a, const Rect& b) {
+     return !(a.right  < b.left  ||
+              a.left   > b.right ||
+              a.bottom < b.top   ||
+              a.top    > b.bottom);
+ }

Key detail: Iterating by auto c (value) would copy each coin, so c.collect() would never modify the actual vector element. Always use auto&.
